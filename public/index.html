require('dotenv').config(); const express = require('express'); const cors = require('cors'); const path = require('path'); const fs = require('fs'); const multer = require('multer'); const pdf = require('pdf-parse'); const OpenAI = require('openai'); const { Pinecone } = require('@pinecone-database/pinecone'); const { v4: uuidv4 } = require('uuid'); const app = express(); const PORT = process.env.PORT || 3000; const upload = multer({ storage: multer.memoryStorage() }); const openai = new OpenAI({ apiKey: process.env.OPENAI_API_KEY }); let pinecone = null; let pineconeIndex = null; const initializePinecone = async () => { try { if (!process.env.PINECONE_API_KEY || !process.env.PINECONE_ENVIRONMENT || !process.env.PINECONE_INDEX) { console.log('⚠ Pinecone not configured. RAG features disabled until env vars are set.'); return; } pinecone = new Pinecone({ apiKey: process.env.PINECONE_API_KEY }); pineconeIndex = pinecone.Index(process.env.PINECONE_INDEX); console.log('✓ Pinecone connected'); } catch (error) { console.error('⚠ Pinecone connection failed:', error.message); } }; initializePinecone(); app.use(cors()); app.use(express.json({ limit: '50mb' })); app.use(express.static(path.join(__dirname, 'public'))); const SOLAR_SYSTEM_PROMPT = `You are a professional and friendly solar energy customer support specialist for ${process.env.COMPANY_NAME || 'SunPeak Solar'}. Your expertise includes: - Residential and commercial solar installations - Solar panel systems, inverters, and monitoring - Battery storage and backup power solutions - Solar financing, leases, and purchase options - Maintenance, repairs, and warranty information - Energy efficiency and solar incentives - Installation timelines and permits Key guidelines: 1. Always be friendly, professional, and helpful. 2. Provide accurate information based on the company knowledge base. 3. When customers ask for quotes or pricing, encourage them to submit a consultation request. 4. For technical issues, suggest troubleshooting steps or recommend a service visit. 5. Mention financing, tax credits, and incentives when relevant. 6. Keep responses concise and easy to read. 7. If you don't know something, be honest and offer to connect them with a specialist. 8. Offer next steps such as a quote request, consultation, or service request. 9. Remind customers they can contact the company at ${process.env.COMPANY_EMAIL || 'support@sunpeaksolar.com'} or ${process.env.COMPANY_PHONE || '+1-555-0123'}. Company Contact Info: - Email: ${process.env.COMPANY_EMAIL || 'support@sunpeaksolar.com'} - Phone: ${process.env.COMPANY_PHONE || '+1-555-0123'} - Website: ${process.env.COMPANY_WEBSITE || 'https://sunpeaksolar.com'} `; const conversationHistory = {}; function getConversationHistory(sessionId) { if (!conversationHistory[sessionId]) conversationHistory[sessionId] = []; return conversationHistory[sessionId]; } function chunkText(text, chunkSize = 1200) { const chunks = []; for (let i = 0; i < text.length; i += chunkSize) { const slice = text.slice(i, i + chunkSize); if (slice.trim()) chunks.push(slice.trim()); } return chunks; } async function searchKnowledge(query) { if (!pineconeIndex) return []; try { const embedding = await openai.embeddings.create({ model: 'text-embedding-3-small', input: query }); const vector = embedding.data[0].embedding; const results = await pineconeIndex.query({ vector, topK: 3, includeMetadata: true }); return results.matches .filter(match => match.score > 0.55) .map(match => match.metadata.content || match.metadata.text) .filter(Boolean); } catch (error) { console.error('Knowledge search failed:', error.message); return []; } } async function embedAndStoreText({ text, sourceName }) { if (!pineconeIndex) { throw new Error('Pinecone is not configured. Add PINECONE_API_KEY, PINECONE_ENVIRONMENT, and PINECONE_INDEX to your .env file.'); } const chunks = chunkText(text, 1200); if (!chunks.length) { throw new Error('No readable text found in the uploaded PDF.'); } const vectors = []; for (let i = 0; i < chunks.length; i += 1) { const chunk = chunks[i]; const embedding = await openai.embeddings.create({ model: 'text-embedding-3-small', input: chunk }); vectors.push({ id: uuidv4(), values: embedding.data[0].embedding, metadata: { source: sourceName, content: chunk, chunkIndex: i, length: chunk.length } }); } await pineconeIndex.upsert(vectors); return { chunks: chunks.length, sourceName }; } async function extractTextFromPdf(buffer, fileName) { const data = await pdf(buffer); const text = (data.text || '').replace(/\s+/g, ' ').trim(); if (!text) { throw new Error(`No text could be extracted from ${fileName}. Please upload a PDF with selectable text.`); } return text; } async function getChatResponse(message, sessionId) { const history = getConversationHistory(sessionId); const relevantDocs = await searchKnowledge(message); let knowledgeContext = ''; if (relevantDocs.length > 0) { knowledgeContext = `\n\nRelevant company information:\n${relevantDocs.map((doc, index) => `${index + 1}. ${doc.substring(0, 350)}...`).join('\n\n')}`; } const systemPrompt = SOLAR_SYSTEM_PROMPT + knowledgeContext; const response = await openai.chat.completions.create({ model: process.env.OPENAI_MODEL || 'gpt-4o-mini', messages: [ { role: 'system', content: systemPrompt }, ...history, { role: 'user', content: message } ], temperature: 0.7, max_tokens: 500 }); const botMessage = response.choices[0].message.content; history.push({ role: 'user', content: message }); history.push({ role: 'assistant', content: botMessage }); if (history.length > 20) { history.splice(0, 2); } return botMessage; } app.get('/health', (req, res) => { const openaiConfigured = !!process.env.OPENAI_API_KEY; const pineconeConfigured = !!pineconeIndex; res.json({ status: 'ok', app: 'solar-chatbot-api-v2', openai_configured: openaiConfigured, pinecone_configured: pineconeConfigured, mode: process.env.NODE_ENV || 'development' }); }); app.get('/api/faqs', (req, res) => { const faqs = [ { question: 'How much does a solar panel system cost?', answer: 'Solar system cost depends on roof size, energy usage, and whether you need battery storage. We can provide a custom quote after a site assessment.' }, { question: 'Do you offer financing?', answer: 'Yes, we offer financing, leases, and cash purchase options based on your needs and budget.' }, { question: 'How long does installation take?', answer: 'Most residential installations are completed in 1-3 days, depending on the system size and permitting timeline.' }, { question: 'Do you provide battery backup systems?', answer: 'Yes. We offer battery solutions for backup power and peak-time savings.' } ]; res.json({ faqs }); }); app.post('/api/chat', async (req, res) => { const { message, customerName, sessionId } = req.body || {}; if (!message || message.trim().length === 0) { return res.status(400).json({ status: 'error', message: 'Message cannot be empty.' }); } const sId = sessionId || `visitor_${Date.now()}_${Math.random().toString(36).substr(2, 9)}`; try { const reply = await getChatResponse(message, sId); res.json({ status: 'success', customerName: customerName || 'Customer', reply, sessionId: sId, timestamp: new Date().toISOString() }); } catch (error) { res.status(500).json({ status: 'error', message: error.message || 'Failed to process your message.' }); } }); app.post('/api/upload-pdf', upload.single('file'), async (req, res) => { try { if (!req.file) { return res.status(400).json({ status: 'error', message: 'Please upload a PDF file.' }); } if (!req.file.originalname.toLowerCase().endsWith('.pdf')) { return res.status(400).json({ status: 'error', message: 'Only PDF files are allowed.' }); } const text = await extractTextFromPdf(req.file.buffer, req.file.originalname); const result = await embedAndStoreText({ text, sourceName: req.file.originalname }); res.json({ status: 'success', message: 'PDF uploaded and indexed successfully.', source: req.file.originalname, chunks: result.chunks }); } catch (error) { console.error('PDF upload failed:', error.message); res.status(500).json({ status: 'error', message: error.message || 'Failed to process the uploaded PDF.' }); } }); app.post('/api/lead', (req, res) => { const { name, email, phone, service, address, roofArea, monthlyBill } = req.body || {}; if (!name || !email) { return res.status(400).json({ status: 'error', message: 'Name and email are required.' }); } const lead = { id: `lead_${Date.now()}`, name, email, phone: phone || 'Not provided', service: service || 'general_inquiry', address: address || 'Not provided', roofArea: roofArea || 'Not provided', monthlyBill: monthlyBill || 'Not provided', timestamp: new Date().toISOString(), source: 'website_chatbot' }; console.log('📋 New lead received:', lead); res.json({ status: 'success', message: 'Thank you for your inquiry! Our team will contact you within 24 hours.', lead }); }); app.get('/api/kb-status', async (req, res) => { if (!pineconeIndex) { return res.json({ status: 'disconnected', message: 'Pinecone is not configured.' }); } try { const stats = await pineconeIndex.describeIndexStats(); res.json({ status: 'connected', totalVectors: stats.totalVectorCount, namespaces: stats.namespaces }); } catch (error) { res.status(500).json({ status: 'error', message: error.message }); } }); app.get('/api/session/:sessionId/clear', (req, res) => { const { sessionId } = req.params; delete conversationHistory[sessionId]; res.json({ status: 'success', message: 'Conversation history cleared.' }); }); app.get('*', (req, res) => { res.sendFile(path.join(__dirname, 'public', 'index.html')); }); app.listen(PORT, () => { console.log('\n🌞 Solar chatbot API v2.0'); console.log('━━━━━━━━━━━━━━━━━━━━━━━━━━━━━'); console.log(`Running on http://localhost:${PORT}`); console.log(`OpenAI API configured: ${process.env.OPENAI_API_KEY ?
